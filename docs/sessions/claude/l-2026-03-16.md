#### equity_rpx Query - WORKING
```sql
SELECT rm.fullpath, rm.mtime, dt.table_name, r.repo_name
FROM data_census.raw_meta_data rm
JOIN data_census.raw_pattern rp ON rm.pattern_id = rp.raw_pattern_id
JOIN data_census.raw_pattern_enriched_map rpm ON rp.raw_pattern_id = rpm.raw_pattern_id
JOIN data_census.delta_table_enriched_pattern_map dtem ON rpm.raw_enriched_id = dtem.raw_enriched_id
JOIN data_census.delta_table dt ON dtem.delta_table_id = dt.table_id
JOIN data_census.repo_delta_table_map rdm ON dt.table_id = rdm.table_id
JOIN data_census.dataset_repo r ON rdm.repo_id = r.repo_id
WHERE dt.table_name = 'equity_rpx' AND rm.year = '2026' AND rm.month = '03'
```
